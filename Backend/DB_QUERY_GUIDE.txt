Step 3: Run SQL queries at the psql prompt

    -- Total resume count
    SELECT COUNT(*) FROM candidate_profile;

    -- Count by date parsed
    SELECT DATE(parsed_at), COUNT(*)
    FROM candidate_profile
    GROUP BY DATE(parsed_at)
    ORDER BY 1 DESC;

    -- Count by user who uploaded
    SELECT uploaded_by, COUNT(*)
    FROM candidate_profile
    GROUP BY uploaded_by
    ORDER BY 2 DESC;

--------------------------------------------------------------------
USEFUL QUERIES
--------------------------------------------------------------------

-- Total resumes
SELECT COUNT(*) AS total FROM candidate_profile;

-- Resumes per day (last 10 days)
SELECT DATE(parsed_at) AS date, COUNT(*) AS count
FROM candidate_profile
WHERE parsed_at IS NOT NULL
GROUP BY DATE(parsed_at)
ORDER BY 1 DESC
LIMIT 10;

-- Resumes uploaded by each user
SELECT COALESCE(uploaded_by, 'legacy/unknown') AS uploaded_by,
       COUNT(*) AS total
FROM candidate_profile
GROUP BY uploaded_by
ORDER BY 2 DESC;

-- Resumes with missing name or email
SELECT COUNT(*) AS missing_name FROM candidate_profile WHERE name IS NULL OR name = '';
SELECT COUNT(*) AS missing_email FROM candidate_profile WHERE email IS NULL OR email = '';

-- Resumes added in the last 7 days
SELECT COUNT(*) FROM candidate_profile
WHERE parsed_at >= NOW() - INTERVAL '7 days';
