│   2. PRIORITIZE BY SOURCE:
│      Explicit marker (80% confidence) > Bold text (70%) > Work history (60%)

│ 4. FIX EXPERIENCE YEARS — Parse Dates Instead of Keyword Matching
├─────────────────────────────────────────────────────────────────────────────┤
│
│ PROBLEM:
│   • Current: Only extracts if resume says "X years of experience" verbatim
│   • Fails on "2015–2024" (should be ~9 years)
│   • Can't extract from implicit dates in work history
│   • Returns NULL or 0 for most resumes
│
│ CURRENT CODE (parser.py):
│   • Regex: r'(\d+)\s+years?\s+of\s+experience'
│   • Very limited pattern matching
│
│ SOLUTION:
│   1. EXTRACT DATE RANGES from work history:
│      • "Jan 2015 – Aug 2024" → calculate 9 years
│      • "2018-2022" → calculate 4 years
│      • "2010 – Present" → calculate current years
│      
│   2. CALCULATE from all work entries:
│      • Sum up all job durations
│      • Result is "Total Experience"
│      
│   3. FALLBACK: Keep keyword matching as backup
│
│ IMPLEMENTATION STEPS:
│   
│   import re
│   from datetime import datetime
│   
│   def extract_experience_years(text: str) -> int:
│       """Extract years of experience from dates or keywords."""
│       
│       # PASS 1: Try keyword match first (most explicit)
│       match = re.search(r'(\d+)\s+years?\s+of\s+experience', text, re.IGNORECASE)
│       if match:
│           return int(match.group(1))
│       
│       # PASS 2: Extract from work history dates
│       total_years = _calculate_from_work_dates(text)
│       if total_years > 0:
│           return total_years
│       
│       # PASS 3: If only one date range exists, use that
│       single_years = _extract_single_date_range(text)
│       if single_years > 0:
│           return single_years
│       
│       return None  # Better NULL than wrong data
│   
│   def _calculate_from_work_dates(text: str) -> int:
│       """Sum all work duration from found date ranges."""
│       # Find patterns like "Jan 2015 – Aug 2024", "2018-2022"
│       patterns = [
│           r'(Jan|Feb|Mar|Apr|May|Jun|Jul|Aug|Sep|Oct|Nov|Dec)\.?\s+(\d{4})\s*[-–]\s*(Jan|Feb|Mar|Apr|May|Jun|Jul|Aug|Sep|Oct|Nov|Dec|Present)\.?\s*(\d{4})?',
│           r'(\d{4})\s*[-–]\s*(\d{4})',  # "2015-2024"
│           r'(\d{4})\s*[-–]\s*(Present|Current)',  # "2015-Present"
│       ]
│       
│       total_months = 0
│       today = datetime.now()
│       
│       for pattern in patterns:
│           for match in re.finditer(pattern, text, re.IGNORECASE):
│               start_year = None
│               end_year = None
│               
│               if len(match.groups()) == 4:
│                   # Full date: "Jan 2015 – Aug 2024"
│                   start_year = int(match.group(2))
│                   end_text = match.group(3)
│                   end_year = int(match.group(4)) if match.group(4) else today.year
│               else:
│                   # Year only: "2015-2024"
│                   start_year = int(match.group(1))
│                   end_year_str = match.group(2)
│                   end_year = today.year if end_year_str.lower() in ['present', 'current'] else int(end_year_str)
│               
│               if start_year and end_year:
│                   total_months += (end_year - start_year) * 12
│       
│       return max(0, total_months // 12)  # Convert to years
│
│
│ EXPECTED IMPACT:
│   📈 Fix 40-50% of experience errors
│   📈 Most reliable improvement (dates are explicit)
│
│ EFFORT: 2-3 hours
│ RISK: Low (only affects numeric extraction)

│      • "DP-900" → "Microsoft Azure Data Fundamentals"

│   3. EXTRACT FULL CERTIFICATION LINE:
│      • "AWS Solutions Architect – Associate" (not just "AWS")

│ IMPLEMENTATION STEPS:
│   
│   from difflib import SequenceMatcher
│   
│   EXAM_CODE_MAPPING = {
│       "AZ-900": "Microsoft Azure Fundamentals",
│       "AZ-104": "Microsoft Azure Administrator",
│       "DP-900": "Microsoft Azure Data Fundamentals",
│       "AZ-305": "Microsoft Azure Solutions Architect",
│       "AWS-SAA": "AWS Certified Solutions Architect – Associate",
│       "AWS-DVA": "AWS Certified Developer – Associate",
│       "AWS-SAP": "AWS Certified Solutions Architect – Professional",
│   }
│   
│   def extract_certifications(text: str) -> list[str]:
│       """Extract certifications with fuzzy matching and exam code lookup."""
│       certs = []
│       
│       # PASS 1: Exact keyword match (highest confidence)
│       for cert_keyword in CERT_KEYWORDS:
│           if cert_keyword.lower() in text.lower():
│               # Extract full line containing keyword
│               line = _extract_line_containing(text, cert_keyword)
│               if line:
│                   certs.append(line)
│       
│       # PASS 2: Fuzzy match (misspellings)
│       words = text.split()
│       for word in words:
│           for cert in CERT_KEYWORDS:
│               similarity = SequenceMatcher(None, word.lower(), cert.lower()).ratio()
│               if similarity > 0.85:  # Match threshold
│                   line = _extract_line_containing(text, word)
│                   if line and line not in certs:
│                       certs.append(line)
│       
│       # PASS 3: Exam code lookup
│       for exam_code, cert_name in EXAM_CODE_MAPPING.items():
│           if exam_code.lower() in text.lower():
│               certs.append(cert_name)
│       
│       # Deduplicate
│       return list(dict.fromkeys(certs))
│

┌─────────────────────────────────────────────────────────────────────────────┐
│ 6. IMPLEMENT SECTION-AWARE EXTRACTION
├─────────────────────────────────────────────────────────────────────────────┤
│
│ CONCEPT:
│   Standard resume sections:
│   • HEADER: Name, email, phone, location, LinkedIn
│   • OBJECTIVE/SUMMARY: What job/role candidate seeks
│   • WORK EXPERIENCE: Companies, titles, dates, descriptions
│   • EDUCATION: Degrees, universities, graduation dates
│   • SKILLS: Technical + soft skills, organized by category
│   • CERTIFICATIONS: Professional certs + exam codes
│   • PROJECTS: Personal/open-source projects
│
│ BENEFIT:
│   • Extract name ONLY from HEADER (not from "John Smith" in recommendations)
│   • Extract education ONLY from EDUCATION section
│   • Extract job titles ONLY from WORK EXPERIENCE
│   • Higher accuracy + fewer false positives
│
│ IMPLEMENTATION:
│   1. Detect section headers via regex patterns
│   2. Split resume into sections using section boundaries
│   3. Run field-specific extraction on each section
│   4. Combine results with source tracking
│
│ EXAMPLE:
│   
│   def detect_resume_sections(text: str) -> dict[str, str]:
│       """Detect and split resume into sections."""
│       section_patterns = {
│           'header': r'^(?:.*?)(?=Objective|Summary|Experience|Education|Skills)',
│           'objective': r'(?:Objective|Summary|Professional Summary)[:\s]*(.+?)(?=Experience|Education|Skills)',
│           'experience': r'(?:Experience|Work History|Professional Experience)[:\s]*(.+?)(?=Education|Skills|Certifications)',
│           'education': r'(?:Education|Academic Background)[:\s]*(.+?)(?=Skills|Certifications|Projects)',
│           'skills': r'(?:Skills|Technical Skills)[:\s]*(.+?)(?=Certifications|Projects)',
│           'certifications': r'(?:Certifications|Certifications & Licenses)[:\s]*(.+?)(?=Projects|$)',
│       }
│       
│       sections = {}
│       for section_name, pattern in section_patterns.items():
│           match = re.search(pattern, text, re.IGNORECASE | re.DOTALL)
│           if match:
│               sections[section_name] = match.group(1).strip()
│       
│       return sections
│   
│   # Then use sections in extraction:
│   sections = detect_resume_sections(resume_text)
│   name = extract_name(sections.get('header', ''))  # Only from header
│   education = extract_education(sections.get('education', ''))  # Only from education
│   skills = extract_skills(sections.get('skills', ''))  # Only from skills
│
│ EFFORT: 4–8 hours
│ EXPECTED IMPACT: 20–30% overall accuracy improvement
│
└─────────────────────────────────────────────────────────────────────────────┘


┌─────────────────────────────────────────────────────────────────────────────┐
│ 7. USE SPACY NER MORE AGGRESSIVELY
├─────────────────────────────────────────────────────────────────────────────┤
│
│ CURRENT USE:
│   • validation_layer.py validation AFTER extraction (too late)
│   • Only checks: name, location, degree
│
│ BETTER APPROACH:
│   • Use spaCy DURING extraction as primary method
│   • Hybrid: regex + spaCy with confidence comparison
│   • spaCy entities available:
│     - PERSON: names
│     - ORG: company/university names
│     - GPE: locations (cities, countries, states)
│     - DATE: dates for experience calculation
│     - PRODUCT: software/tool names
│
│ EXAMPLE:
│   
│   def extract_name_hybrid(text: str) -> tuple[str, str]:
│       """Extract name using REGEX + spaCy with confidence ranking."""
│       
│       # PASS 1: Regex extraction (fast)
│       regex_name = _extract_name_regex(text[:300])
│       regex_confidence = _score_name_regex(regex_name)
│       
│       # PASS 2: spaCy NER extraction (accurate)
│       if _SPACY_AVAILABLE:
│           doc = _NLP(text[:300])git 
│           spacy_persons = [ent.text for ent in doc.ents if ent.label_ == "PERSON"]
│           if spacy_persons:
│               spacy_name = spacy_persons[0]
│               spacy_confidence = 0.95  # spaCy tends to be accurate
│               
│               # Compare confidence
│               if spacy_confidence > regex_confidence:
│                   return spacy_name, "spacy"
│       
│       return regex_name, "regex"
│
│ EFFORT: 3–4 hours
│ EXPECTED IMPACT: 10–15% improvement (especially names, locations)
│
└─────────────────────────────────────────────────────────────────────────────┘


┌─────────────────────────────────────────────────────────────────────────────┐
│ 8. ADD FIELD-LEVEL CONFIDENCE TRACKING
├─────────────────────────────────────────────────────────────────────────────┤
│
│ CURRENT ISSUE:
│   • Global confidence score (all-or-nothing)
│   • If name confidence low, ENTIRE resume sent to LLM
│   • Wastes API calls on fields that are already correct
│
│ SOLUTION:
│   • Calculate confidence PER FIELD
│   • Only invoke LLM for fields with confidence < threshold
│   • Track confidence in database for manual review
│
│ EXAMPLE:
│   
│   extracted = {
│       'name': ('John Smith', 0.92, 'regex'),
│       'location': ('Hyderabad', 0.65, 'regex'),  # LOW confidence
│       'job_title': ('Senior Developer', 0.88, 'regex'),
│       'years_exp': (None, 0.0, 'none'),  # NOT extracted
│       'certifications': ('AWS, Azure', 0.75, 'regex'),
│   }
│   
│   # Only enhance low-confidence fields
│   fields_to_enhance = {k: v for k, v in extracted.items() 
│                        if v[1] < 0.75}
│   
│   enhanced = hybrid_enhancer.enhance(fields_to_enhance)
│   
│   # Save confidence scores
│   store_in_db(candidate_id, extracted, enhanced)
│
│ BENEFIT:
│   • 60–70% fewer LLM API calls
│   • Better audit trail (know which fields are weak)
│   • Can flag for manual review
│
│ EFFORT: 2–3 hours
│ EXPECTED IMPACT: Cost reduction + better data quality tracking
│

def run_tests():
    """Test parser accuracy on sample resumes."""
    TEST_RESUMES = [
        {
            "filename": "john_smith.pdf", 
            "expected": {
                "name": "John Smith", 
                "location": "NYC",
                "job_title": "Senior Developer"
            }
        },
        {
            "filename": "divya_sharma.pdf", 
            "expected": {
                "name": "Divya Sharma", 
                "location": "Bangalore",
                "job_title": "Data Engineer"
            }
        },
        # ... 100+ cases
    ]
    
    results = {"passed": 0, "failed": 0}
    
    for test in TEST_RESUMES:
        try:
            extracted = parser.parse_file(test["filename"])
            
            for field, expected_value in test["expected"].items():
                actual_value = extracted.get(field)
                if actual_value == expected_value:
                    results["passed"] += 1
                else:
                    results["failed"] += 1
                    print(f"FAIL: {test['filename']} / {field}")
                    print(f"  Expected: {expected_value}")
                    print(f"  Got: {actual_value}")
        except Exception as e:
            results["failed"] += 1
            print(f"ERROR: {test['filename']}: {e}")
    
    accuracy = results["passed"] / (results["passed"] + results["failed"])
    print(f"\nOverall Accuracy: {accuracy:.1%}")
    return results
