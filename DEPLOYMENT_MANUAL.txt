==============================================================
  ROLLBACK PROCEDURE
==============================================================

If production deployment breaks something:

--- Quick Rollback (revert to previous Docker image) ----------
   ssh -i "C:\Users\prave\.ssh\id_ed25519" root@89.167.60.41
   cd /root/resume-parser
   git revert HEAD --no-edit
   git push origin prod
   # Then trigger GitHub Actions or run deploy_prod.sh

--- Database Rollback -----------------------------------------
   # List available backups
   ls /root/resume-parser/database_backups/

   # Restore a backup
   docker exec -i resume-db-prod psql -U postgres -d postgres -c "DROP DATABASE resume_prod;"
   docker exec -i resume-db-prod psql -U postgres -d postgres -c "CREATE DATABASE resume_prod;"
   docker exec -i resume-db-prod psql -U postgres -d resume_prod \
     < /root/resume-parser/database_backups/production_backup_YYYYMMDD_HHMMSS.sql

  -f docker-compose.yml -f docker-compose.prod.yml \
  up -d --build api ingestion

docker exec -it resume-db-dev psql -U postgres -d resume_dev

  "DELETE FROM candidate_skills_profile; DELETE FROM candidate_profile; \
   UPDATE users SET resumes_uploaded = 0; \
