# Example: How to use google_drive_sync in your own parser

from pathlib import Path
from google_drive_sync import sync_drive_folder

# Configuration
GDRIVE_FOLDER_ID = "your-folder-id-here"
CREDENTIALS_JSON = Path("path/to/credentials.json")
TOKEN_JSON = Path(".gdrive_token.json")
DOWNLOAD_DIR = Path("resumes_cache")

# Sync files from Google Drive
scanned, downloaded = sync_drive_folder(
    folder_id=GDRIVE_FOLDER_ID,
    download_dir=DOWNLOAD_DIR,
    credentials_json=CREDENTIALS_JSON,
    token_json=TOKEN_JSON,
    allowed_exts={".pdf", ".docx"}  # Only these file types
)

print(f"Scanned {scanned} files, downloaded {downloaded} new/updated files")

# Now process the files in DOWNLOAD_DIR with YOUR parser
for resume_file in DOWNLOAD_DIR.glob("*.pdf"):
    # Call your own parsing function
    your_parse_function(resume_file)

for resume_file in DOWNLOAD_DIR.glob("*.docx"):
    # Call your own parsing function
    your_parse_function(resume_file)


================================================================================
FUNCTION SIGNATURE
================================================================================

def sync_drive_folder(
    *,
    folder_id: str,              # Google Drive folder ID
    download_dir: Path,          # Where to download files
    credentials_json: Path,      # Path to credentials JSON
    token_json: Path,            # Path to token cache
    allowed_exts: set[str] | None = None  # e.g., {".pdf", ".docx"}
) -> tuple[int, int]:  # Returns (scanned_count, downloaded_count)
    """
    Syncs a Google Drive folder to local directory.
    Only downloads new or modified files (uses MD5 checksum).
    Returns tuple of (files_scanned, files_downloaded)
    """

AUTHENTICATION TYPES SUPPORTED
================================================================================

1. SERVICE ACCOUNT (Recommended for automation)
   - Non-interactive
   - Works for background processes
   - Credentials file looks like:
     {
       "type": "service_account",
       "project_id": "...",
       "private_key_id": "...",
       ...
     }
   - Share Drive folder with service account email

2. OAUTH USER CREDENTIALS (For interactive use)
   - First run opens browser for authentication
   - Subsequent runs use cached token
   - Credentials file is OAuth client secrets JSON

================================================================================
SMART CACHING BEHAVIOR
================================================================================

The sync_drive_folder function:
1. Lists all files in the Google Drive folder
2. Compares with local files using MD5 checksum
3. ONLY downloads files that are:
   - New (not in local cache)
   - Modified (different MD5 checksum)
4. Skips unchanged files completely
5. Handles filename collisions by appending drive file ID

This means:
- First run: Downloads all files
- Subsequent runs: Only downloads new/changed files
- Very efficient for large resume collections

# your_resume_parser.py

import os
from pathlib import Path
from dotenv import load_dotenv
from google_drive_sync import sync_drive_folder

# Load configuration
load_dotenv()

def parse_resume(file_path):
    """Your own resume parsing logic here"""
    print(f"Parsing: {file_path.name}")
    # Your parsing code...
    # Extract name, email, skills, etc.
    # Store in your database
    print(f"Processed: {file_path.name}")

def main():
    # Google Drive configuration
    folder_id = os.getenv("GDRIVE_FOLDER_ID")
    creds_json = Path(os.getenv("GDRIVE_CREDENTIALS_JSON", "credentials.json"))
    token_json = Path(".gdrive_token.json")
    cache_dir = Path("resumes_cache")
    
    if not folder_id:
        print("Error: GDRIVE_FOLDER_ID not set")
        return 1
    
    if not creds_json.exists():
        print(f"Error: Credentials file not found: {creds_json}")
        return 1
    
    print("Syncing from Google Drive...")
    scanned, downloaded = sync_drive_folder(
        folder_id=folder_id,
        download_dir=cache_dir,
        credentials_json=creds_json,
        token_json=token_json,
        allowed_exts={".pdf", ".docx"}
    )
    print(f"Sync complete: {scanned} scanned, {downloaded} downloaded\n")
    
    # Process all resumes in cache directory
    print("Processing resumes...")
    resume_files = list(cache_dir.glob("*.pdf")) + list(cache_dir.glob("*.docx"))
    
    for resume_file in resume_files:
        try:
            parse_resume(resume_file)
        except Exception as e:
            print(f"Error parsing {resume_file.name}: {e}")
    
    print(f"\nProcessed {len(resume_files)} resumes")
    return 0

if __name__ == "__main__":
    exit(main())

================================================================================
TROUBLESHOOTING
================================================================================

Issue: "Module not found: google_drive_sync"
→ Make sure google_drive_sync.py is in same directory or PYTHONPATH

Issue: "No module named 'googleapiclient'"
→ Install dependencies: pip install -r requirements_gdrive.txt

Issue: "Authentication failed"
→ Check credentials.json is valid
→ Verify folder is shared with service account email

Issue: "No files downloaded"
→ Check GDRIVE_FOLDER_ID is correct
→ Verify files exist in the Drive folder
→ Check allowed_exts includes your file types

Issue: "Permission denied"
→ Share the Drive folder with service account email
→ Or use OAuth credentials with proper permissions
